// Gov AI chat
var govHistory=[];
function saveGovKey(){
  var k=document.getElementById('govKey').value.trim();
  if(k.startsWith('sk-ant-')){
    localStorage.setItem('avidaa_anthropic_key',k);
    document.getElementById('govKeyArea').style.display='none';
    addGovMsg('sistema','Klave wardá. Bo por hasi preguntanan tokante gobernashon di Kòrsou.');
  } else {
    addGovMsg('sistema','Klave no validó. Mester kuminsá ku sk-ant-');
  }
}
function forgetGovKey(){
  localStorage.removeItem('avidaa_anthropic_key');
  govHistory=[];
  document.getElementById('govMsgs').innerHTML='';
  document.getElementById('govKeyArea').style.display='block';
}
function addGovMsg(role,text){
  var el=document.getElementById('govMsgs');
  var msg=document.createElement('div');
  msg.style.cssText='padding:10px 14px;margin-bottom:8px;font-size:15px;line-height:1.75;'+(role==='user'?'color:var(--gold2);text-align:right;':'color:rgba(245,237,224,.82);background:rgba(0,200,190,.04);border-left:2px solid var(--border);');
  msg.textContent=text;
  el.appendChild(msg);
  el.scrollTop=el.scrollHeight;
}
function sendGov(){
  var key=localStorage.getItem('avidaa_anthropic_key');
  if(!key){addGovMsg('sistema','Pone bo klave promé.');return;}
  var input=document.getElementById('govInput');
  var q=input.value.trim();
  if(!q)return;
  input.value='';
  addGovMsg('user',q);
  govHistory.push({role:'user',content:q});
  addGovMsg('ai','...');
  var msgs=document.getElementById('govMsgs');
  var typing=msgs.lastChild;
  fetch('https://api.anthropic.com/v1/messages',{
    method:'POST',
    headers:{'Content-Type':'application/json','x-api-key':key,'anthropic-version':'2023-06-01','anthropic-dangerous-direct-browser-access':'true'},
    body:JSON.stringify({
      model:'claude-sonnet-4-6',
      max_tokens:600,
      system:'Bo ta un asistente edukativo riba gobernashon di Kòrsou. Kontestá SOLAMENTE na Papiamentu di Kòrsou (fonemiko, no Aruban Papiamento). Fokó riba edukashon siviko: ministerionan, lei, demokrasia, struktura di gobernashon, Reino di Hulanda. Nunka menshoná nòmber di politician, no duna opinion politiko, no kompartí data personal. Si pregunta riba nòmber di un minister, eksplíká ku e sekshon aki ta edukashon siviko solamente. Keep answers clear and educational.',
      messages:govHistory
    })
  }).then(function(r){return r.json();}).then(function(d){
    var reply=d.content&&d.content[0]?d.content[0].text:'No por kontestá awor.';
    msgs.removeChild(typing);
    addGovMsg('ai',reply);
    govHistory.push({role:'assistant',content:reply});
  }).catch(function(){msgs.removeChild(typing);addGovMsg('ai','Error di konekshon. Purba atrobe.');});
}

    if(el)el.style.display='none';
  }
})();
